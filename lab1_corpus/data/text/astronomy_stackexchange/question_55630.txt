def transform_data(pos, vel):
pos = coord.CartesianRepresentation(pos.x, pos.y, 0 * u.au)
vel = coord.CartesianDifferential(vel.x, vel.y, 0 * u.au / u.s)
vel = vel.norm().to(u.m / u.s).value
dist = pos.norm().to(u.AU).value
return dist, vel
