The main one up front is this script is supposed to get Stellarium to load with specific parameters, take a screen shot, and presumably exit or at least return control to the calling script. Stellarium comes up with most of the specified options, but never takes a screenshot, and does not exit so the script can create another screen short with the latitude variable incremented by 1.
#! /usr/bin/python3
import os
import subprocess
import time
# Set the path to Stellarium and FFmpeg
stellarium_path = "/usr/bin/stellarium" # Adjust as necessary
ffmpeg_path = "/usr/bin/ffmpeg" # Adjust as necessary
# Create a directory to store images
output_dir = "/RAID/Server-Main/FlatEarth/Stellarium Screen Shots/Video"
os.makedirs(output_dir, exist_ok=True)
def capture_images(object_name, field_of_view, start_latitude, end_latitude, longitude, date, dtime):
# Loop through the specified latitude range
for latitude in range(start_latitude, end_latitude + 1):
# Prepare the Stellarium command
command = f'"{stellarium_path}" --full-screen=no --script=screenshot_script.py --longitude={longitude} --latitude={latitude} --fov={field_of_view} --object={object_name} --screenshot-dir="{output_dir}" --sky-date={date} --sky-time={dtime}'
# Execute the Stellarium command
print(f"Capturing image at Latitude: {latitude}")
subprocess.run(command, shell=True)
# Wait for a moment to ensure Stellarium has time to process
time.sleep(5)
def create_video(output_dir):
# Create a video from the captured images
video_output = "output_video.mp4"
ffmpeg_command = f'"{ffmpeg_path}" -framerate 1 -i "{output_dir}/image_lat%%d.png" -c:v libx264 -pix_fmt yuv420p "{video_output}"'
print("Creating video...")
print(f"ffmpeg command: {ffmpeg_command}")
subprocess.run(ffmpeg_command, shell=True)
print(f"Video created: {video_output}")
if __name__ == "__main__":
# User-defined parameters

create_video(output_dir)
