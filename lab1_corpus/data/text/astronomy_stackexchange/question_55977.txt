def ICRStoGal(ra,dec):
coord = ra + dec
c_icrs = SkyCoord((coord), unit=(u.hourangle, u.deg))
c_gal =c_icrs.galactic
l,b=c_gal.l.degree, c_gal.b.degree
return l,b
