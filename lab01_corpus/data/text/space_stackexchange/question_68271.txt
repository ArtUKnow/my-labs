rebound + SPICE (scipy) simulation produces highly inclined lunar orbit
I am using the python rebound library with SPICE data to simulate the solar system.
Below is a full script which simulates the solar system for 1 year beginning at 2000, and here is the moon_rel.csv output which is the position of the moon relative to earth in km, sampled once per day.
If I chart the moon position, the results are odd.
Some aspects of this are plausible -- in the XY plane the mean distance is about right. From this I deduce that the XY plane is approximately the ecliptic.
However the lunar orbit is highly inclined from the XY plane, clearly much more than is correct.

Full source:
import rebound
import spiceypy as spice
import numpy as np
import os
import sys
# Load SPICE kernels
spice.furnsh("naif0012.tls") # Leap second kernel
spice.furnsh("de440s.bsp") # Planetary ephemerides
def get_state(body, epoch):
# Get position and velocity (km and km/s) from ephemeris
state, _ = spice.spkezr(body, epoch, "J2000", "NONE", "SSB")
position = np.array(state[:3]) * 1e3 # convert km to meters
velocity = np.array(state[3:]) * 1e3 # convert km/s to m/s
return position, velocity
# Masses in kg (from NASA fact sheets)
mass_dict = {
"SUN": 1.98847e30,
"MERCURY": 3.3011e23,
"VENUS": 4.8675e24,
"EARTH": 5.9722e24,
"MOON": 7.342e22,
"MARS BARYCENTER": 6.4171e23,
"JUPITER BARYCENTER": 1.8982e27,
"SATURN BARYCENTER": 5.6834e26,
"URANUS BARYCENTER": 8.6810e25,
"NEPTUNE BARYCENTER": 1.02413e26,
}
# Create simulation
sim = rebound.Simulation()
sim.G = 6.67430e-11 # m^3 / kg / s^2
# J2000 epoch
et = spice.str2et("2000-01-01T12:00:00")
# Add bodies
for body in mass_dict.keys():
pos, vel = get_state(body, et)
m = mass_dict[body]
sim.add(m=m, x=pos[0], y=pos[1], z=pos[2],
vx=vel[0], vy=vel[1], vz=vel[2],
hash=body)
sim.move_to_com()
sim.integrator = "WHFast"
sim.dt = 60*60 # 1 hour per step
earth_abs = []
moon_abs = []
moon_rel = []
t = 0
for day in range(365):
t += 60*60*24 # 1 day per loop
sim.integrate(t)
moon = sim.particles["MOON"]
earth = sim.particles["EARTH"]
m_xyz = np.array([moon.x, moon.y, moon.z])
e_xyz = np.array([earth.x, earth.y, earth.z])
earth_abs.append(e_xyz/1.496e+11) # AU
moon_abs.append(m_xyz/1.496e+11) # AU
moon_rel.append((m_xyz-e_xyz)/1000) # km
def to_csv(values):
return ",".join([str(x) for x in values]) + "\n"
def dump_csv(table, filename):
with open(filename, "w") as f:
for row in table:
f.write(to_csv(row))
dump_csv(earth_abs,'/tmp/earth_abs.csv')
dump_csv(moon_abs,'/tmp/moon_abs.csv')
dump_csv(moon_rel,'/tmp/moon_rel.csv')
