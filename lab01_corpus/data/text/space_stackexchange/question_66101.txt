import numpy as np
import os
# Get the directory of the current script
marsatm_dir = os.path.dirname(os.path.abspath(__file__))
# Path to the marsatm file
marsatm_file_path = os.path.join(marsatm_dir, "marsatm.txt")
def marsinit():
data_list = [] #create an empty list to store the data for each height
with open(marsatm_file_path, 'r') as file:
for line in file:
height, temp, rho, a = line.strip().split() #extract the data from the file and assign relevant variables
data_list.append([float(height), float(temp), float(rho), float(a)]) #store the variables as a list in the data list
data_array = np.array(data_list) #create a variable that defines the array, where each list in data_list becomes a row in the array (aka matrix)
return data_array
if __name__ == '__main__': #if the name of the code that is being run = marsatm, this section will run, otherwise it will be skipped.
while True: #ensure that the input is valid
h = int(input(f"height in m:")) / 1000
if h > 150 or h < 0:
print("Input is out of range, please enter a kilometre value between 0 and 150.")
continue
break
def marsatm (h, data_array):
for i in range(len(data_array[:,0])): #this loops iterates the process, of finding for which i (row in matrix) it is true, that h lies between two rows.
if data_array[i,0] <= h <= data_array[i+1,0]:
k = (h - data_array [ i,0 ]) / (data_array [ i + 1 , 0] - data_array [ i,0 ]) #computing the k value
t = ( 1 - k ) * data_array [ i , 1 ] + k * data_array [ i + 1 , 1 ] #computing the temperature
d = ( 1 - k ) * data_array [ i , 2 ] + k * data_array [ i + 1 , 2 ] #computing the density
a = ( 1 - k ) * data_array [ i , 3 ] + k * data_array [ i + 1 , 3 ] #computing the speed of sound
return t, d, a #return a tuple containing (t,d,a)

$$
\Delta V = \Delta V_{thrust} + \Delta V_{drag} - \Delta V_{gravity}
