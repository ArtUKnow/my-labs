def steps_n(val: float) -> int:
if val < 1.0:
return 0
steps, curr = 0, val
for _ in range(20):
nxt = np.floor(curr / LAMBDA_Y)
if nxt < 1.0:
break
curr, steps = nxt, steps + 1

return r_eff, S_n, S_k, S_h
